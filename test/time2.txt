data=rrr.txt
export TIMEFORMAT=%lU

echo -ne "perl print none\t"
time perl -ane 'BEGIN { <>; } $x++;' < $data > /dev/null

echo -ne "perl print none\t"
time perl -ane 'BEGIN { <>; } $x++;' < $data > /dev/null

echo -ne "perl print one\t"
time perl -ane 'BEGIN { $"="\t"; } print "$F[0]\n";' < $data > /dev/null

echo -ne "perl print all\t"
time perl -ane 'BEGIN { $"="\t"; } print "@F\n";' < $data > /dev/null

echo -ne "perl print all, add column (addition)\t"
time perl -ane 'BEGIN { $"="\t"; } push @F, $F[0] + $F[1]; print "@F\n";' < $data > /dev/null

echo -ne "$prog print none\t"
time $prog -c < $data > /dev/null

echo -ne "$prog print one\t"
time $prog one < $data > /dev/null

echo -ne "$prog print all\t"
time $prog -A < $data > /dev/null

echo -ne "$prog print all, twice\t"
time $prog -A < $data | $prog -A > /dev/null

echo -ne "$prog print all, thrice\t"
time $prog -A < $data | $prog -A | $prog -A > /dev/null

echo -ne "$prog print all, add column (addition)\t"
time $prog -A < $data ::one:two,add > /dev/null

echo -ne "$prog print all plus compute\t"
time $prog -A foo:=one:two,add < $data > /dev/null

echo -ne "$prog print all plus long compute\t"
time $prog -A foo:=one:two:three:four:five,add,add,add,add < $data > /dev/null

echo -ne "$prog print all plus long compute shortcut\t"
time $prog -A foo:=one:two:three:four:five,addall < $data > /dev/null
