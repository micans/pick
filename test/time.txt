data=rrr.txt
export TIMEFORMAT="%lU"

# >&2 echo "-- timing ${prog?Need prog to be set} --"

echo -ne "perl one comparison\t"
time perl -ane 'BEGIN { <>; } END { print "$x\n" } $x++ if $F[0] > $F[1];' < $data > /dev/null

echo -ne "perl one comparison\t"
time perl -ane 'BEGIN { <>; } END { print "$x\n" } $x++ if $F[0] > $F[1];' < $data > /dev/null

echo -ne "perl two comparisons\t"
time perl -ane 'BEGIN { <>; } END { print "$x\n" } $x++ if $F[0] > $F[1] && $F[2] > $F[3];' < $data > /dev/null

echo -ne "$prog one comparison\t"
time $prog -cq @one/gt/:two < $data > /dev/null

echo -ne "$prog two comparisons\t"
time $prog -cq @one/gt/:two @three/gt/:four < $data > /dev/null

echo -ne "$prog one compute (addition)\t"
time $prog -c onep:=one^1,add < $data > /dev/null

echo -ne "$prog two computes (addition)\t"
time $prog -c onep:=one^1,add twop:=two^1,add < $data > /dev/null

echo -ne "$prog three computes (addition)\t"
time $prog -c onep:=one^1,add twop:=two^1,add thrp:=three^1,add < $data > /dev/null

echo -ne "$prog four computes (addition)\t"
time $prog -c onep:=one^1,add twop:=two^1,add thrp:=three^1,add fourp:=four^1,add < $data > /dev/null

echo -ne "$prog five computes (addition)\t"
time $prog -c onep:=one^1,add twop:=two^1,add thrp:=three^1,add fourp:=four^1,add fivep:=five^1,add < $data > /dev/null

echo -ne "$prog five computes (multiplication)\t"
time $prog -c onep:=one^13,mul twop:=two^17,mul thrp:=three^19,mul fourp:=four^23,mul fivep:=five^29,mul < $data > /dev/null

echo -ne "$prog one compute, (five add operators)\t"
time $prog -c onep:=one^1,add^1,add^1,add^1,add^1,add < $data > /dev/null

echo -ne "$prog three hash lookups (32k keys)\t"
time $prog -c --fdict-EN=num.txt.gz  ::,rowno one:two:three/sfx::__^EN,map,catall,len < rrr.txt > /dev/null

echo -ne "$prog three hash lookups (10 keys)\t"
time $prog -c --fdict-DIGIT=zeronine.txt  ::,rowno one:two:three/sfx::__^10,mod^DIGIT,map < rrr.txt > /dev/null
